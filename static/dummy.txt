import json
from typing import List, Set, Optional, Dict
from collections import defaultdict, Counter
from datetime import date, timedelta
import random
import pandas as pd

# --- Flatten JSON with global exclusion ---
def flatten_json_exclude(y, exclude_suffixes: Optional[List[str]] = None, parent_key='', sep='.') -> Dict[str, any]:
    exclude_suffixes = exclude_suffixes or []
    items = []

    if isinstance(y, dict):
        for k, v in y.items():
            new_key = f"{parent_key}{sep}{k}" if parent_key else k
            if any(new_key.endswith(suffix) for suffix in exclude_suffixes):
                continue
            items.extend(flatten_json_exclude(v, exclude_suffixes, new_key, sep=sep).items())
    elif isinstance(y, list):
        for i, v in enumerate(y):
            new_key = f"{parent_key}{sep}{i}" if parent_key else str(i)
            items.extend(flatten_json_exclude(v, exclude_suffixes, new_key, sep=sep).items())
    else:
        if not any(parent_key.endswith(suffix) for suffix in exclude_suffixes):
            items.append((parent_key, y))
    return dict(items)

# --- Field Summary Generator ---
def summarize_fields_for_user(docs, exclude_suffixes=None):
    field_stats = defaultdict(lambda: {
        "count": 0,
        "examples": Counter(),
        "types": Counter()
    })

    for doc in docs:
        flat = flatten_json_exclude(doc, exclude_suffixes=exclude_suffixes)
        for key, value in flat.items():
            field_stats[key]["count"] += 1
            field_stats[key]["examples"][str(value)] += 1
            field_stats[key]["types"][type(value).__name__] += 1

    summary = []
    total_docs = len(docs)
    for field, stat in field_stats.items():
        summary.append({
            "field": field,
            "type": stat["types"].most_common(1)[0][0],
            "doc_coverage": stat["count"],
            "coverage_percent": round(100 * stat["count"] / total_docs, 1),
            "unique_values": len(stat["examples"]),
            "examples": ", ".join(list(stat["examples"].keys())[:3]),
            "enum_like": len(stat["examples"]) <= 10
        })

    return pd.DataFrame(summary).sort_values(by="coverage_percent", ascending=False)

# --- Match user sampling fields to actual flattened keys (with inclusion) ---
def match_suffix_fields_with_inclusion(flat_doc, user_fields: List[str], include_suffixes: Optional[List[str]] = None) -> dict:
    matched = {}
    for user_field in user_fields:
        for k in flat_doc.keys():
            if k.endswith(user_field):
                if include_suffixes is None or any(k.endswith(suf) for suf in include_suffixes):
                    matched[user_field] = k
                    break
    return matched

# --- Suffix-based Sampling with Inclusion + Exclusion ---
def sample_by_field_suffix_combination_with_inclusion(
    docs: List[dict],
    sampling_fields: List[str],
    include_suffixes: Optional[List[str]] = None,
    exclude_suffixes: Optional[List[str]] = None
) -> List[dict]:
    seen_combinations = set()
    selected_docs = []

    for doc in docs:
        flat = flatten_json_exclude(doc, exclude_suffixes)
        matched_fields = match_suffix_fields_with_inclusion(flat, sampling_fields, include_suffixes)
        combo = tuple(flat.get(matched_fields.get(field, ""), None) for field in sampling_fields)
        if combo not in seen_combinations:
            selected_docs.append(doc)
            seen_combinations.add(combo)

    return selected_docs

# --- Synthetic demo data generator ---
def generate_synthetic_docs(n=50):
    base_date = date(2025, 3, 1)
    docs = []
    for _ in range(n):
        trips = random.randint(1, 5)
        doc = {"A": {"B": {}, "X": {}}}
        for i in range(1, trips + 1):
            d = (base_date + timedelta(days=random.randint(0, 20))).isoformat()
            t = random.choice(["car", "public_transport", "bike", "walk"])
            doc["A"]["B"][str(i)] = d
            doc["A"]["X"][str(i)] = t
            doc["meta"] = {"session_id": random.randint(1000, 2000)}
        docs.append(doc)
    return docs

# --- MAIN: Example usage ---
if __name__ == "__main__":
    # 1. Load or generate data
    docs = generate_synthetic_docs()

    # 2. Define field rules
    exclude_suffixes = ["meta.session_id"]
    include_suffixes = ["A.B.1", "A.X.1"]
    sampling_fields = ["A.B.1", "A.X.1"]

    # 3. Field summary (for inspection)
    summary_df = summarize_fields_for_user(docs, exclude_suffixes=exclude_suffixes)
    summary_df.to_csv("field_summary.csv", index=False)
    print("Saved field summary to field_summary.csv")

    # 4. Sampling by suffix with inclusion
    sampled_docs = sample_by_field_suffix_combination_with_inclusion(
        docs,
        sampling_fields=sampling_fields,
        include_suffixes=include_suffixes,
        exclude_suffixes=exclude_suffixes
    )

    # 5. Output
    with open("sampled_output.json", "w") as f:
        json.dump(sampled_docs, f, indent=2)
    print(f"Saved {len(sampled_docs)} documents to sampled_output.json")
