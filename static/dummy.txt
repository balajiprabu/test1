import json
from typing import List, Set, Optional, Dict
from collections import defaultdict, Counter
from datetime import date, timedelta
import random
import pandas as pd

# --- 1. Flatten JSON with exclusion ---
def flatten_json_exclude(y, exclude_suffixes: Optional[List[str]] = None, parent_key='', sep='.') -> Dict[str, any]:
    exclude_suffixes = exclude_suffixes or []
    items = []

    if isinstance(y, dict):
        for k, v in y.items():
            new_key = f"{parent_key}{sep}{k}" if parent_key else k
            if any(new_key.endswith(suffix) for suffix in exclude_suffixes):
                continue
            items.extend(flatten_json_exclude(v, exclude_suffixes, new_key, sep=sep).items())
    elif isinstance(y, list):
        for i, v in enumerate(y):
            new_key = f"{parent_key}{sep}{i}" if parent_key else str(i)
            items.extend(flatten_json_exclude(v, exclude_suffixes, new_key, sep=sep).items())
    else:
        if not any(parent_key.endswith(suffix) for suffix in exclude_suffixes):
            items.append((parent_key, y))
    return dict(items)

# --- 2. Field summary ---
def summarize_fields_for_user(docs, exclude_suffixes=None):
    field_stats = defaultdict(lambda: {
        "count": 0,
        "examples": Counter(),
        "types": Counter()
    })

    for doc in docs:
        flat = flatten_json_exclude(doc, exclude_suffixes=exclude_suffixes)
        for key, value in flat.items():
            field_stats[key]["count"] += 1
            field_stats[key]["examples"][str(value)] += 1
            field_stats[key]["types"][type(value).__name__] += 1

    summary = []
    total_docs = len(docs)
    for field, stat in field_stats.items():
        summary.append({
            "field": field,
            "type": stat["types"].most_common(1)[0][0],
            "doc_coverage": stat["count"],
            "coverage_percent": round(100 * stat["count"] / total_docs, 1),
            "unique_values": len(stat["examples"]),
            "examples": ", ".join(list(stat["examples"].keys())[:3]),
            "enum_like": len(stat["examples"]) <= 10
        })

    return pd.DataFrame(summary).sort_values(by="coverage_percent", ascending=False)

# --- 3. Sampling by field suffixes ---
def match_suffix_fields(flat_doc, user_fields: List[str]) -> dict:
    matched = {}
    for user_field in user_fields:
        for k in flat_doc.keys():
            if k.endswith(user_field):
                matched[user_field] = k
                break
    return matched

def sample_by_field_suffix_combination(docs: List[dict], sampling_fields: List[str], exclude_suffixes=None) -> List[dict]:
    seen_combinations = set()
    selected_docs = []

    for doc in docs:
        flat = flatten_json_exclude(doc, exclude_suffixes)
        matched_fields = match_suffix_fields(flat, sampling_fields)
        combo = tuple(flat.get(matched_fields.get(field, ""), None) for field in sampling_fields)
        if combo not in seen_combinations:
            selected_docs.append(doc)
            seen_combinations.add(combo)

    return selected_docs

# --- 4. Sampling by synthetic field pairs ---
def extract_synthetic_pairs(doc: dict, left_prefix: str, right_prefix: str) -> Set[str]:
    synthetic_pairs = set()
    flat = flatten_json_exclude(doc)

    left_map = {}
    right_map = {}

    for k, v in flat.items():
        if k.startswith(left_prefix + ".") and k[len(left_prefix) + 1:].isdigit():
            idx = k.split(".")[-1]
            left_map[idx] = v
        elif k.startswith(right_prefix + ".") and k[len(right_prefix) + 1:].isdigit():
            idx = k.split(".")[-1]
            right_map[idx] = v

    for idx in set(left_map.keys()).intersection(right_map.keys()):
        synthetic_pairs.add(f"{str(right_map[idx])}@{str(left_map[idx])}")

    return synthetic_pairs

def sample_by_parallel_pairs(docs: List[dict], left_prefix: str, right_prefix: str) -> List[dict]:
    all_pairs = {}
    for i, doc in enumerate(docs):
        pairs = extract_synthetic_pairs(doc, left_prefix, right_prefix)
        for p in pairs:
            all_pairs.setdefault(p, set()).add(i)

    uncovered = set(all_pairs.keys())
    selected_docs = []
    used_docs = set()

    while uncovered:
        best_doc = None
        best_coverage = set()

        for i, doc in enumerate(docs):
            if i in used_docs:
                continue
            doc_pairs = extract_synthetic_pairs(doc, left_prefix, right_prefix)
            coverage = uncovered.intersection(doc_pairs)
            if len(coverage) > len(best_coverage):
                best_doc = i
                best_coverage = coverage

        if best_doc is not None:
            selected_docs.append(docs[best_doc])
            used_docs.add(best_doc)
            uncovered -= best_coverage
        else:
            break

    return selected_docs

# --- 5. Example usage with synthetic data ---
def generate_synthetic_docs(n=50):
    base_date = date(2025, 3, 1)
    docs = []
    for _ in range(n):
        trips = random.randint(1, 5)
        doc = {"A": {"B": {}, "X": {}}}
        for i in range(1, trips + 1):
            d = (base_date + timedelta(days=random.randint(0, 20))).isoformat()
            t = random.choice(["car", "public_transport", "bike", "walk"])
            doc["A"]["B"][str(i)] = d
            doc["A"]["X"][str(i)] = t
            doc["meta"] = {"session_id": random.randint(1000, 2000)}
        docs.append(doc)
    return docs

    exclude_fields = ["meta.session_id"]
    summary_df = summarize_fields_for_user(docs, exclude_suffixes=exclude_fields)
    summary_df.to_csv("field_summary.csv", index=False)
    print("Saved field summary to field_summary.csv")

    # Step 2: Choose one mode — update these based on what you want

    # Option A: Sample by suffix fields
    suffix_fields = ["A.X.1", "A.B.1"]  # user-selected fields
    suffix_samples = sample_by_field_suffix_combination(docs, suffix_fields, exclude_suffixes=exclude_fields)
    with open("samples_by_suffix.json", "w") as f:
        json.dump(suffix_samples, f, indent=2)
    print(f"Saved {len(suffix_samples)} suffix-based samples to samples_by_suffix.json")

    # Option B: Sample by synthetic field pair (e.g., transport@date)
    pair_samples = sample_by_parallel_pairs(docs, left_prefix="A.B", right_prefix="A.X")
    with open("samples_by_pair.json", "w") as f:
        json.dump(pair_samples, f, indent=2)
    print(f"Saved {len(pair_samples)} synthetic-pair-based samples to samples_by_pair.json")
